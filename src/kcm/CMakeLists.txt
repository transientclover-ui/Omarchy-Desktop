cmake_minimum_required(VERSION 3.16)
project(frankenstein_kcm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_library(kcm_frankenstein_shell MODULE shell.cpp)
target_link_libraries(kcm_frankenstein_shell PRIVATE Qt6::Widgets KF6::KCMUtils KF6::CoreAddons)
install(TARGETS kcm_frankenstein_shell LIBRARY DESTINATION lib/qt6/plugins/plasma/kcms/systemsettings)
install(FILES kcm_frankenstein_shell.desktop DESTINATION share/applications)
set_target_properties(kcm_frankenstein_shell PROPERTIES PREFIX "")
